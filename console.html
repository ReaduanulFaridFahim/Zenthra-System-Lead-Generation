    function onEmailableFilterChange() {
      const search = (document.getElementById('em-search-input')?.value || '').trim().toLowerCase();
      const status = document.getElementById('em-status-select')?.value || 'READY';
      const country = document.getElementById('em-country-select')?.value || 'ALL';
      const service = document.getElementById('em-service-select')?.value || 'ALL';

      emailableFilteredContacts = emailableContactsData.filter(c => {
        // Status filter
        if (status === 'READY' && c.status !== 'READY_TO_EMAIL') return false;
        if (status === 'SENT' && c.status !== 'EMAIL_SENT') return false;
        if (status === 'CLEARED' && c.status !== 'CLEARED') return false;
        if (status !== 'CLEARED' && c.status === 'CLEARED') return false;

        // Country filter
        if (country !== 'ALL' && c.country !== country) return false;

        // Service filter
        if (service !== 'ALL') {
          const sLower = service.toLowerCase();
          const wedge = (c.recommended_wedge || '').toLowerCase();
          const svc = (c.offered_service || '').toLowerCase();
          if (!wedge.includes(sLower) && !svc.includes(sLower)) return false;
        }

        // Search query
        if (search) {
          const hay = `${c.name} ${c.first_name} ${c.last_name} ${c.company_name} ${c.email} ${c.job_title} ${c.country} ${c.domain} ${c.recommended_wedge}`.toLowerCase();
          if (!hay.includes(search)) return false;
        }

        return true;
      });

      renderEmailableTable(emailableFilteredContacts);
    }

    function toggleSelectAllEmailable(masterCheck) {
      const isChecked = masterCheck.checked;

    function updateBatchButtonLabel() {
      const btn = document.getElementById('btn-batch-oneclick');

      const deleteBtn = document.getElementById('btn-clear-selected-contacts');
      const countSpan = document.getElementById('clear-sel-count');

    async function clearSelectedEmailableContacts() {
      const count = selectedContactIds.size;
      if (count === 0) return;
      if (!confirm(`Are you sure you want to clear and remove ${count} selected contact(s) from your "Who Can I Email" roster?`)) return;

      try {
        const res = await fetch('/api/contacts/clear', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ contact_ids: Array.from(selectedContactIds) })
        });
        const data = await res.json();
        if (data.success) {
          alert(`✅ Cleared ${data.count} contact(s) from your active roster.`);
          selectedContactIds.clear();
          updateBatchButtonLabel();
          loadEmailableContacts();
        } else {
          alert('Failed to clear contacts: ' + (data.error || 'Unknown error'));
        }
      } catch (err) {
        alert('Error: ' + err.message);
      }
    }

    async function sendOneClickMail(contactId) {
      const contact = emailableContactsData.find(c => c.contact_id === contactId);
      if (!contact) {
        alert('Contact record not found.');
        return;
      }

      const btn = document.getElementById(`btn-1click-${contactId}`);
      if (btn) {
        btn.textContent = '⚡ Sending via SMTP...';
        btn.disabled = true;
      }

      try {
        const res = await fetch('/api/send-one-click-mail', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({
            contact_id: contact.contact_id,
            account_id: contact.account_id,
            recipient_email: contact.email,
            recipient_name: contact.name,
            company_name: contact.company_name,
            service_wedge: contact.recommended_wedge,
            deal_value_usd: contact.deal_value_usd || 8500,
            subject: contact.draft_subject,
            body: contact.draft_body
          })
        });

        const data = await res.json();

        if (data.success) {
          contact.status = 'EMAIL_SENT';
          contact.last_sent_at = new Date().toISOString();
          alert(`✓ 1-Click Email successfully dispatched via connect.zenthra@gmail.com to ${contact.name} <${contact.email}>!\n\nMessage ID: ${data.message_id || 'Accepted by SMTP'}\nDeal staged in CRM: $${(contact.deal_value_usd || 8500).toLocaleString()} USD`);
          loadEmailableContacts();
          fetchPlatformData();
        } else {
          alert('Failed to send 1-click email: ' + (data.error || 'Unknown error'));
          if (btn) {
            btn.textContent = '⚡ 1-Click Send';
            btn.disabled = false;
          }
        }
      } catch (err) {
        alert('Send request error: ' + err.message);
        if (btn) {
          btn.textContent = '⚡ 1-Click Send';
          btn.disabled = false;
        }
      }
    }

    async function triggerBatchOneClickSend() {
      let targets = [];
      if (selectedContactIds.size > 0) {
        targets = emailableContactsData.filter(c => selectedContactIds.has(c.contact_id) && c.status === 'READY_TO_EMAIL');
      } else {
        targets = emailableFilteredContacts.filter(c => c.status === 'READY_TO_EMAIL');
      }

      if (targets.length === 0) {
        alert('No ready-to-email contacts found to dispatch. Ensure selected contacts have status READY.');
        return;
      }

      const count = Math.min(targets.length, 25);
      const confirmed = confirm(`Are you sure you want to 1-click dispatch personalized proposal emails to ${count} verified decision-maker(s) via active Gmail SMTP (connect.zenthra@gmail.com)?\n\nThis will send real emails, stage deals in the CRM, and trigger live n8n webhooks.`);
      if (!confirmed) return;

      const btn = document.getElementById('btn-batch-oneclick');
      if (btn) {
        btn.textContent = `🚀 Dispatching ${count} via SMTP...`;
        btn.disabled = true;
      }

      try {
        const res = await fetch('/api/batch-send-one-click-mail', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({
            contact_ids: targets.slice(0, count).map(c => c.contact_id)
          })
        });

        const data = await res.json();

        if (data.success) {
          selectedContactIds.clear();
          alert(`✓ Batch outreach complete! Successfully dispatched ${data.sent_count} emails via Gmail SMTP.\n\nAll deals are staged in the SQLite CRM and n8n webhooks have been triggered.`);
          loadEmailableContacts();
          fetchPlatformData();
        } else {
          alert('Batch dispatch error: ' + (data.error || 'Server error'));
        }
      } catch (err) {
        alert('Batch dispatch network error: ' + err.message);
      } finally {
        if (btn) {
          btn.textContent = '🚀 1-Click Send All Ready Emails';
          btn.disabled = false;
        }
      }
    }
